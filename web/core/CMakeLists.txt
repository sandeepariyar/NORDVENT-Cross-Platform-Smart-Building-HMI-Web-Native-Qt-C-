cmake_minimum_required(VERSION 3.16)
project(NordventCore LANGUAGES CXX)
add_library(nordvent_core STATIC src/NordventCore.cpp)
target_include_directories(nordvent_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(nordvent_core PUBLIC cxx_std_17)
add_executable(nordvent_core_tests tests/core_tests.cpp)
target_link_libraries(nordvent_core_tests PRIVATE nordvent_core)
enable_testing()
add_test(NAME nordvent_core_tests COMMAND nordvent_core_tests)
